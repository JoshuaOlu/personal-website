{%- comment -%} Prints one publication as a reference, in the style Authors (Year). Title. Venue, details. https://doi.org/... Your own name (listed in _data/authors.yml) is printed in bold. Usage: {% include citation.html pub=pub %} {%- endcomment -%} {%- assign p = include.pub -%} {%- assign shown_authors = p.authors | escape -%} {%- assign already_bold = false -%} {%- for name in site.data.authors.highlight -%} {%- if already_bold == false and shown_authors contains name -%} {%- assign bold_name = "" | append: name | append: "" -%} {%- assign shown_authors = shown_authors | replace: name, bold_name -%} {%- assign already_bold = true -%} {%- endif -%} {%- endfor -%} {{ shown_authors }} ({{ p.year }}{% if p.month %}, {{ p.month }}{% endif %}). {{ p.title | escape }}. {% if p.prefix %}{{ p.prefix | escape }} {% endif %}{{ p.venue | escape }}{% if p.details %}, {{ p.details | escape }}{% endif %}. {%- if p.doi %} https://doi.org/{{ p.doi }} {%- elsif p.url %} {{ p.url }}{% endif -%}
